require_relative 'peep'

peep = Peep.new(File.binread('song.flac'))      # WAV or FLAC, from memory
puts peep.rate, peep.channels, peep.total
loop do
  samples = peep.read(1024)                    # Array of interleaved stereo int16
  break if samples.empty?
  play(samples)                                # your audio output
end
peep.seek(44_100)                              # jump to frame 44100
peep.gain = 128                                # half volume
peep.close
